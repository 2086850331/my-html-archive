TB_ReaderRecords 相关查询整理

1. 查询未离开的刷卡记录（可改为更新离开时间）

筛选设备类别为 2、离开时间为空、读取时间早于指定日期的记录
select *--update t set F_LeaveTime=getdate()
from TB_ReaderRecords t 
where F_DeviceCategory = 2 
  and F_LeaveTime is null 
  and F_ReadTime < '2026-09-21' 
order by F_ReadTime desc

2. 设备 / 安全事件 / 阅读器 状态关联查询

查看设备的基本信息、阅读器信息、最新刷卡记录时间
--设备/安全事件/阅读器 状态关联查询
SELECT 
    d.F_DeviceCode,
    d.F_DeviceName,
    d.F_RFIDReaderId,
    r.F_RFIDName,
    r.F_RFIDReader,
    CASE d.F_EnabledMark
        WHEN 0 THEN '禁用'
        WHEN 1 THEN '启用'
        ELSE '未知' 
    END AS [启用状态],
    CASE d.F_UseState
        WHEN 0 THEN '未使用'
        WHEN 1 THEN '使用中'
        WHEN 2 THEN '安全'
        WHEN 3 THEN '異常'
        ELSE '未知' 
    END AS [使用状态],
    r.F_LastCheckStateTime AS [阅读器最后数据同步时间],
    rec.LatestReadTime AS [最新记录时间(F_ReadTime)]
FROM TB_Device d
LEFT JOIN TB_RFIDReader r 
    ON d.F_RFIDReaderId = r.F_RFIDReaderId
LEFT JOIN (
    SELECT 
        F_DeviceId, 
        MAX(F_ReadTime) AS LatestReadTime
    FROM TB_ReaderRecords
    GROUP BY F_DeviceId
) rec 
    ON d.F_DeviceId = rec.F_DeviceId
WHERE d.F_DeviceCategory = 2 AND d.F_DeleteMark = 0
ORDER BY rec.LatestReadTime DESC;

3. 阅读器“在线 / 离线”及是否上传记录查询

统计每个阅读器的在线状态、关联设备、历史上传记录总数及最后上传时间
--阅读器“在线”“离线” 是否上传记录查询
SELECT 
    r.F_RFIDName AS [阅读器名称],
    r.F_RFIDReader AS [阅读器编号],
    CASE r.F_OnLineState 
        WHEN 1 THEN '在线' 
        WHEN 0 THEN '离线' 
        ELSE '未知' 
    END AS [当前状态],
    d.F_DeviceCode,
    d.F_DeviceName,
    CASE 
        WHEN MAX(t.F_ReadTime) IS NULL THEN '从未上传' 
        ELSE '已上传' 
    END AS [是否上传记录],
    COUNT(t.F_ReaderId) AS [历史上传记录总数], 
    MAX(t.F_ReadTime) AS [最后一次上传时间]
FROM TB_RFIDReader r
LEFT JOIN TB_Device d 
    ON r.F_RFIDReaderId = d.F_RFIDReaderId 
    AND d.F_DeleteMark = 0
LEFT JOIN TB_ReaderRecords t 
    ON d.F_DeviceId = t.F_DeviceId 
WHERE r.F_DeleteMark = 0
GROUP BY 
    r.F_RFIDReaderId,
    r.F_RFIDName,
    r.F_RFIDReader,
    r.F_OnLineState,
    d.F_DeviceCode,
    d.F_DeviceName
ORDER BY 
    r.F_OnLineState DESC,
    [最后一次上传时间] DESC;
